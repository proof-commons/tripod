%   ways, and the rows below keep the cases where it forced a BLUNT name and rejected a

%   SYMMETRY ROWS (the renames that prove this is precision):
%     R -> phi        RENAMED.  R overclaims ("return"); the object is Omega/Y, monotone
%                     non-decreasing — a FLOOR.  phi underclaims and is therefore truer.
%     B -> A  (map)   RENAMED.  B reads "burn/balance"; the exported object is a

%     pi -> kappa     RENAMED.  pi collides with the constant and reads "market price";
%                     mandated APPLIED kappa(d) keeps it visibly a price-of-something.
%     Y               KEPT.  Y is accurate as supply.  We forbid GLOSSING it "yield";
%                     we do not rename it — renaming an accurate symbol would be optics.
%     burn            KEPT (operation).  Bluntest, most checkable name for "destroy own
%                     units irrecoverably".  Its connotations point TOWARD scrutiny, not
%                     away; the operation is never renamed after its output.
%     deposit         KEPT (operation).  The warehouse-deposit reading is the accurate
%                     one under the receipt model; we do not flee it to "contribute".
%     G -> SP         RENAMED on TWO sufficient grounds, each stated: (1) precision — the
%                     family labels PROVEN PROPERTIES of a constructed object, not
%                     assurances to a counterparty; "specified property" is the exact
%                     reading.  (2) collision-safety — two-letter no-hyphen SP is mutually
%                     unambiguous against P- (pins), D (residuals), T (rules), I (invariant).
%                     Either ground alone suffices.  Neither ground is a category avoided.
%     TITLE = ATTESTATION   NAMED FOR THE EXPORT.  "Token" (the original working title)
%                     named the SCAFFOLDING (the burnable receipt, the consumed input);
%                     the specification exports exactly
%                     ONE object, the attestation map A, and "everything else is internal
%                     structure that makes the specified properties hold."  The title now
%                     agrees with the export A, not with the container Y that A outlives.
%                     This finishes the B->A rename at the document level.  The INSTRUMENT

%                     what it PRODUCES, not for what it DESTROYS.
%     PREFIX = A-   NAMED on the same ground: the prefix names the DOCUMENT, and it
%                     follows the document's name.
%

% Net live share: the post-fee, immediately-redeemable share of a bootstrapping-phase mint,
% per reserve unit deposited = (1−f)·ζ.  A PURPOSE-NEUTRAL allocation ratio: it asserts no
% depositor intent.  A reader producing attestation reads its RECIPROCAL as the reserve cost
% of one attestation; a reader holding the position reads it as liquid retention.  This layer
% asserts neither projection (cf. the locality discipline: as the layer declines to PRICE the
% attestation, it declines to NAME the depositor's purpose).  FULL-FORM by design (renders the
% product, no new glyph) so the value cannot be misstated.  Phase scope is part of the export.
% First use in a consumer document must cite def:interface:net-live-share.
\NewDocumentCommand{\netliveshare}{}{ \ensuremath{ (1-\feerate)\,\livesplit } }

% [interface], derived companion for the cost projection (full-form, no new glyph).  The
% reserve cost of one attestation unit = 1/((1−f)ζ).  Provided so the cost PROJECTION renders
% honestly (direction-correct) wherever a consumer wants it, without re-deriving.  NOT a fifth
% interface object — a convenience over \netliveshare; the export count stays four.
\NewDocumentCommand{\attestunitcost}{}{ \ensuremath{ 1/\bigl((1-\feerate)\,\livesplit\bigr) } }

% Destination-addressed burn: burn(x; a) — destroy x live-class units; record on address a
% an attestation of x·φ at the prevailing floor.  Funder unconstrained; consent-free (the
% record is monotone in the address's favor).  x is the ITALIC burn amount.
% KEPT "burn": the bluntest, most checkable name for irrecoverable destruction of one's own
% units; the operation is never renamed after its output.
\cs_new_protected:Npn \tripod_burnop:nn #1#2 { \operatorname{burn}(#1;#2) }
\NewDocumentCommand{\burnop}{m m}{ \ensuremath{ \tripod_burnop:nn {#1}{#2} } }

% Deposit: deposit(δ; a) — δ reserve units in; receipt minted at the next cycle boundary to
% holder address a, split ζ : (1−ζ) into LIVE : TIME-LOCKED during bootstrapping (all LIVE
% in the normal phase), less the mint fee f.
% KEPT "deposit": the warehouse-deposit reading is the accurate one under the receipt model.
\cs_new_protected:Npn \tripod_depositop:nn #1#2 { \operatorname{deposit}(#1;#2) }
\NewDocumentCommand{\depositop}{m m}{ \ensuremath{ \tripod_depositop:nn {#1}{#2} } }

% Redemption: redeem(x) — return x live-class units; receive x·φ reserve units.

\NewDocumentCommand{\redeemop}{m}{ \ensuremath{ \tripod_redeemop:n {#1} } }

% Redemption-rate floor φ = Ω/Y — reserve per receipt unit; the redemption price and THE
